export default {
  name: 'Enemies flee',
  description: 'Enemies turn tail and run away from the player.',
  patchNotes: 'Enemies have discovered fear. They now run away from you with tremendous commitment.',
  hooks: {
    onTick(g, dt) {
      for (const enemy of g.enemies()) {
        const dx = enemy.x - g.player.x;
        const dy = enemy.y - g.player.y;
        const distance = Math.sqrt(dx * dx + dy * dy);
        const nx = distance > 0.001 ? dx / distance : 1;
        const ny = distance > 0.001 ? dy / distance : 0;
        g.set(enemy, { vx: nx * 400, vy: ny * 400 });
      }
    },
  },
};
